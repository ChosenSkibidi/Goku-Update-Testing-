local function IsCharacterPart(part)
    local ancestor = part.Parent
    while ancestor do
        if ancestor:IsA("Model") and ancestor:FindFirstChildOfClass("Humanoid") then
            return true
        end
        ancestor = ancestor.Parent
    end
    return false
end

            end
        end

            if obj:IsA("BasePart") and obj.Material == Enum.Material.Concrete and not IsCharacterPart(obj) then
                ApplyConcreteTexture(obj)
            end
        end)

        warn("[ConcreteReplacer] Workspace.Map folder not found.")
    end

local nextImpactIndex = 1

local function PlayRandomDodgeSound()
    local snd = preloadedDodgeSounds[math.random(#preloadedDodgeSounds)]
    if snd.IsPlaying then snd:Stop() end
    snd:Play()
end

local function PlayNextImpactSound()
    local snd = preloadedImpactSounds[nextImpactIndex]
    if snd.IsPlaying then snd:Stop() end
    snd:Play()
    nextImpactIndex = nextImpactIndex + 1
    if nextImpactIndex > #preloadedImpactSounds then nextImpactIndex = 1 end
end

local lastDodgeAnimId = nil
local blockAnimId = "rbxassetid://10470389827"
local blockLoopActive = false

local function PickRandomDodgeAnim()
    if #dodgeAnimIds == 1 then return dodgeAnimIds[1] end
    local picked
    repeat
        picked = dodgeAnimIds[math.random(#dodgeAnimIds)]
    until picked ~= lastDodgeAnimId
    lastDodgeAnimId = picked
    return picked
end

local function PlayDodgeAnimation()
    local anim = Instance.new("Animation")
    anim.AnimationId = PickRandomDodgeAnim()
    for _, track in animator:GetPlayingAnimationTracks() do
        if table.find(dodgeAnimIds, track.Animation.AnimationId) or track.Animation.AnimationId == blockAnimId then
            track:Stop()
        end
    end
    local track = animator:LoadAnimation(anim)
    track.Priority = Enum.AnimationPriority.Action4
    track:Play()
    track:AdjustSpeed(Random.new():NextNumber(1, 1.5))
    PlayRandomDodgeSound()
    PlayNextImpactSound()
end

local prevBlockReact = character:GetAttribute("BlockReact")

local function CheckBlockReact()
    local isBlocking = character:GetAttribute("Blocking")
    local blockReact  = character:GetAttribute("BlockReact")
    if not blockReact then return end
    if isBlocking then
        if math.abs(math.abs(blockReact) - math.abs(prevBlockReact)) > 1 then
            PlayDodgeAnimation()
        elseif math.abs(blockReact) < 1 and not blockLoopActive then
            local anim = Instance.new("Animation")
            anim.AnimationId = blockAnimId
            local track = animator:LoadAnimation(anim)
            track.Looped = true
            track:Play()
            blockLoopActive = true
        end
    end
    prevBlockReact = blockReact
end

            if table.find(dodgeAnimIds, track.Animation.AnimationId) or track.Animation.AnimationId == blockAnimId then
                track:Stop()
                blockLoopActive = false
            end
        end
    end
end)

local function DownloadSongFile(url, filename)
    if not isfile(filename) then
        writefile(filename, game:HttpGet(url))
    end

    local readOk, contents = pcall(readfile, filename)
    if not readOk or #contents < minValidSongFileBytes then
        writefile(filename, game:HttpGet(url))
    end
end

-- Install record. Every install writes this JSON file:
--   { "count": <how many pack songs are installed>, "installed": [ <their names> ] }
local dbPackManifestFile = "DBMusicPack.json"

-- Returns the recorded names as a set. A missing or unreadable file = nothing recorded.
local function ReadDbPackManifest()
    local recorded = {}
    if not isfile(dbPackManifestFile) then return recorded end

    local readOk, decoded = pcall(function()
        return game:GetService("HttpService"):JSONDecode(readfile(dbPackManifestFile))
    end)
    if readOk and type(decoded) == "table" and type(decoded.installed) == "table" then
        for _, name in ipairs(decoded.installed) do
            recorded[name] = true
        end
    end
    return recorded
end

-- Adds one song to the record and rewrites the JSON (count + names, in pack order).
-- Names whose file has since gone missing are dropped so the record stays truthful.
local function RecordDbPackSongInstalled(songName)
    local recorded = ReadDbPackManifest()
    recorded[songName] = true

    local names = {}
    for _, song in ipairs(dbSongs) do
        if recorded[song.name] and (song.name == songName or isfile(song.file)) then
            table.insert(names, song.name)
        end
    end

    writefile(dbPackManifestFile, game:GetService("HttpService"):JSONEncode({
        count = #names,
        installed = names,
    }))
end

-- Checks the script's pack list against the record. Returns the names that are
-- installed and the songs that still need installing (new songs added to the
-- script, or ones whose file is gone).
local function GetDbPackStatus()
    local recorded = ReadDbPackManifest()
    local installedNames = {}
    local missingSongs = {}
    for _, song in ipairs(dbSongs) do
        if recorded[song.name] and isfile(song.file) then
            table.insert(installedNames, song.name)
        else
            table.insert(missingSongs, song)
        end
    end
    return installedNames, missingSongs
end

-- Downloads one pack song and checks the file is real (a failed download can
-- leave a tiny error page behind - that gets deleted so it can't pass as installed).
local function InstallDbPackSong(song)
    DownloadSongFile(song.url, song.file)
    local readOk, contents = pcall(readfile, song.file)
    if not readOk or #contents < minValidSongFileBytes then
        pcall(delfile, song.file)
        error("downloaded file was invalid")
    end
end

local function WaitForSoundToLoad(sound, timeoutSeconds)
    if sound.IsLoaded then return end
    local isLoaded = false
    local loadedConnection
    loadedConnection = sound.Loaded:Connect(function()
        isLoaded = true
    end)
    local elapsedSeconds = 0
    while not isLoaded and not sound.IsLoaded and elapsedSeconds < timeoutSeconds do
        task.wait(0.1)
        elapsedSeconds = elapsedSeconds + 0.1
    end
    loadedConnection:Disconnect()
end

local musicActive            = true
local currentSongIndex       = nil
local usedStartingSong       = false
local currentMusicSound      = nil
local prematureEndRetryCount = 0
local currentFadeInTween      = nil
local playbackGeneration     = 0

-- Which pack the shuffle draws from: "Regular" (musicSongs) or "DB" (dbSongs).
-- Only one is ever active; picking a song from the other pack's dropdown
-- switches it on and this one off.
local activePack = "Regular"

local function GetSongsForPack(packName)
    return packName == "DB" and dbSongs or musicSongs
end

local function FindSongIndexByName(name)
    local songs = GetSongsForPack(activePack)
    local wanted = name:lower():gsub("^%s+", ""):gsub("%s+$", "")
    -- 1) exact name, 2) exact file name (with or without .mp3), 3) name contains the text
    for i, song in ipairs(songs) do
        if song.name:lower() == wanted then
            return i
        end
    end
    for i, song in ipairs(songs) do
        local fileLower = song.file:lower()
        if fileLower == wanted or fileLower:gsub("%.mp3$", "") == wanted then
            return i
        end
    end
    for i, song in ipairs(songs) do
        if song.name:lower():find(wanted, 1, true) then
            return i
        end
    end
    return nil
end

local function PickNextSongIndex()
    local songCount = #GetSongsForPack(activePack)
    if songCount == 1 then return 1 end
    local index
    repeat
        index = math.random(songCount)
    until index ~= currentSongIndex
    return index
end

local function PlaySoundCycle(isRetryingCurrentSong, forcedSongIndex)
    if not musicActive then return end

    -- Every cycle gets a generation number. If something newer starts (a song
    -- picked from the panel, music toggled off/on) the older cycle sees the
    -- mismatch and stops itself instead of chaining into another random song.
    playbackGeneration = playbackGeneration + 1
    local myGeneration = playbackGeneration

    if forcedSongIndex then
        prematureEndRetryCount = 0
        usedStartingSong = true
        currentSongIndex = forcedSongIndex
    elseif isRetryingCurrentSong then
        prematureEndRetryCount = prematureEndRetryCount + 1
    else
        prematureEndRetryCount = 0
        if not usedStartingSong then
            usedStartingSong = true
            local startIndex = (typeof(_G.StartingSong) == "string" and _G.StartingSong ~= "") and FindSongIndexByName(_G.StartingSong)
            if typeof(_G.StartingSong) == "string" and _G.StartingSong ~= "" and not startIndex then
                warn("[Music] StartingSong \"" .. _G.StartingSong .. "\" didn't match any song, picking randomly")
            end
            currentSongIndex = startIndex or PickNextSongIndex()
        else
            currentSongIndex = PickNextSongIndex()
        end
    end

    local entry = GetSongsForPack(activePack)[currentSongIndex]
    print("[Music] Now playing: " .. entry.name)

    local sound = GetGitSound(entry.url, entry.file)
    sound.Name   = entry.name
    sound.Volume = 0
    currentMusicSound = sound

    WaitForSoundToLoad(sound, 5)
    if not musicActive or not sound.Parent or myGeneration ~= playbackGeneration then
        sound:Destroy()
        return
    end

    sound:Play()
    local songStartClock = os.clock()

    local fadeInTween = TweenService:Create(sound, TweenInfo.new(5, Enum.EasingStyle.Linear), { Volume = musicVolume })
    currentFadeInTween = fadeInTween
    fadeInTween:Play()

    local songFinished = false
    local function FinishSong(reachedEnd)
        if songFinished then return end
        songFinished = true
        sound:Destroy()

        if not musicActive then return end
        if myGeneration ~= playbackGeneration then return end

        if reachedEnd or prematureEndRetryCount >= maxPrematureEndRetries then
            PlaySoundCycle()
        else
            PlaySoundCycle(true)
        end
    end

    sound.Ended:Connect(function()
        local playedFullLength = sound.TimeLength > 0 and (os.clock() - songStartClock) >= sound.TimeLength - 2
        local reachedEnd = sound.TimeLength <= 0 or sound.TimePosition >= sound.TimeLength - 1 or playedFullLength
        FinishSong(reachedEnd)
    end)

    if entry.name:lower():find("lucy~", 1, true) then
        task.spawn(function()
            while not songFinished and musicActive and sound.Parent and sound.TimePosition < LucyFadeOutStart do
                task.wait(0.1)
            end
            if songFinished or not musicActive or not sound.Parent then return end

            local fadeOutDuration = math.max(LucySongCap - LucyFadeOutStart, 0.1)
            local fadeOutTween = TweenService:Create(sound, TweenInfo.new(fadeOutDuration, Enum.EasingStyle.Linear), { Volume = 0 })
            fadeOutTween:Play()

            while not songFinished and musicActive and sound.Parent and sound.TimePosition < LucySongCap do
                task.wait(0.1)
            end
            if songFinished then return end

            sound:Stop()
            FinishSong(true)
        end)
    end
end

-- Immediately swaps whatever is playing for song `index` of the given pack
-- ("Regular" or "DB"; defaults to the active one) and makes that pack the active
-- pack. Used by the panel's song dropdowns.
local function PlaySongAtIndex(index, packName)
    packName = packName or activePack
    if not GetSongsForPack(packName)[index] then return end

    activePack = packName
    playbackGeneration = playbackGeneration + 1

    if currentFadeInTween then
        currentFadeInTween:Cancel()
        currentFadeInTween = nil
    end
    if currentMusicSound then
        currentMusicSound:Stop()
        currentMusicSound:Destroy()
        currentMusicSound = nil
    end

    _G.Music    = true
    musicActive = true
    task.spawn(PlaySoundCycle, false, index)
end

    -- record again: all installed -> the row becomes the song dropdown; anything
    -- still missing -> "Retry" (after a failure) or "N+ Needs to be installed".
    StartDbPackInstall = function()
        if dbPackInstalling then return end
        local songsToInstall = dbPackMissingSongs

        dbPackInstallFailed = false
        dbPackInstallDone = 0

            for _, songEntry in ipairs(songsToInstall) do

                    InstallDbPackSong(songEntry)
                    RecordDbPackSongInstalled(songEntry.name)
                end)
                if not installOk then
                    warn("[DB Music Pack] Couldn't install " .. songEntry.name .. ": " .. tostring(installError))
                    dbPackInstallFailed = true
                    break
                end
                dbPackInstallDone = dbPackInstallDone + 1

local function DoTeleport()
    local char = localPlayer.Character or localPlayer.CharacterAdded:Wait()
    local hrp  = char:WaitForChild("HumanoidRootPart")

    local allTargets = {}

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= localPlayer and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
            table.insert(allTargets, plr.Character.HumanoidRootPart)
        end
    end

    local liveFolder = Workspace:FindFirstChild("Live")
    if liveFolder then
        for _, obj in ipairs(liveFolder:GetChildren()) do
            if obj.Name == "Weakest Dummy" and obj:FindFirstChild("HumanoidRootPart") then
                table.insert(allTargets, obj.HumanoidRootPart)
            end
        end
    end

    local nearestTarget = nil
    local closestDist   = math.huge
    for _, target in ipairs(allTargets) do
        local dist = (target.Position - hrp.Position).Magnitude
        if dist < closestDist then
            closestDist   = dist
            nearestTarget = target
        end
    end

    local originalTransparency = {}
    local function StoreTransparency(obj)
        if obj:IsA("BasePart") or obj:IsA("Decal") or obj:IsA("Texture") then
            originalTransparency[obj] = obj.Transparency
        end
        for _, child in ipairs(obj:GetChildren()) do
            StoreTransparency(child)
        end
    end
    StoreTransparency(char)

    local function SetVisibility(hide)
        for obj, trans in pairs(originalTransparency) do
            if trans < 1 then
                obj.Transparency = hide and 1 or trans
            end
        end
    end

    PlayAnimation("120577018823573", 0, 0.5, 2)
    task.wait(0.5)

    EmitVFX(ReplicatedStorage.Resources.NinjaUlt.Start_Effects.Particles.Low)
    EmitVFX(ReplicatedStorage.Resources.NinjaUlt.Start_Effects.Particles.Center_Tilted)
    EmitVFX(ReplicatedStorage.Resources.NinjaUlt.Start_Effects.Particles.Ground)

    local teleportSfx = Instance.new("Sound")
    teleportSfx.SoundId = "rbxassetid://3339374157"
    teleportSfx.Volume  = 1
    teleportSfx.Parent  = hrp
    teleportSfx:Play()

    if nearestTarget then
        local dir = (nearestTarget.Position - hrp.Position).Unit
        hrp.CFrame = CFrame.new(nearestTarget.Position - dir * -3, nearestTarget.Position)
    end

    SetVisibility(true)  task.wait(0.01)
    SetVisibility(false) task.wait(0.01)
    SetVisibility(true)  task.wait(0.01)
    SetVisibility(false)
end

    ["Consecutive Punches"]    = "Wrenching Ki Blasts",
    ["Shove"]                  = "Spirit Crusher",
    ["Uppercut"]               = "Sky Breaker",
    ["Serious Punch"]          = "?",
    ["Omni Directional Punch"] = "?",
    ["Death Counter"]          = "Divine Counter",
    ["Table Flip"]             = "Spirit Bomb",
}

local function updateToolNames()
    for _, slot in ipairs(hotbar:GetChildren()) do
        if slot:FindFirstChild("Base") and slot.Base:FindFirstChild("ToolName") then
            local toolName = slot.Base.ToolName
            local newName  = toolNameMap[toolName.Text]
            if newName then
                if not slot:FindFirstChild("CustomName") then
                    toolName.Visible = false
                    local clone = toolName:Clone()
                    clone.Name    = "CustomName"
                    clone.Parent  = slot
                    clone.Text    = newName
                    clone.Visible = true
                elseif slot.CustomName.Text ~= newName then
                    toolName.Visible     = false
                    slot.CustomName.Text = newName
                end
            end
        end
    end
end

task.spawn(function()
    while true do
        updateToolNames()
        task.wait(1)
    end
end)

local function RevertSSJBHairToSSJ()
    local oldHair = character:FindFirstChild("GokuSSJHair")
    if not oldHair or not oldHair:GetAttribute("IsSSJB") then return end

    local oldLight = oldHair:FindFirstChildOfClass("PointLight")
    oldHair:Destroy()

    local gokuHairForms = Workspace:FindFirstChild("GokuHairForms")
    local ssjSource = gokuHairForms and gokuHairForms["Super saiyan 1"] and gokuHairForms["Super saiyan 1"]["Ssj dominated (Original)"]
    if not ssjSource then return end

    local newHair      = ssjSource:Clone()
    newHair.Name        = "GokuSSJHair"
    newHair.Anchored    = false
    newHair.CanCollide  = false
    newHair.CFrame      = headPart.CFrame * CFrame.new(0, 1.1, 0.5) * CFrame.Angles(0, math.rad(180), 0)
    newHair.Parent      = character
    WeldToHead(newHair)
    ssjHair = newHair

    if oldLight then
        oldLight.Color  = Color3.fromRGB(255, 255, 150)
        oldLight.Parent = newHair
    end
end

local function DisableSSJBAura()
    for _, obj in ipairs(character:GetDescendants()) do
        if obj:IsA("ParticleEmitter") or obj:IsA("Beam") or obj:IsA("PointLight") or obj:IsA("SpotLight") or obj:IsA("SurfaceLight") then
            if obj:GetAttribute("LimitedAura") or obj:GetAttribute("InnerRageAura") or obj:GetAttribute("LimAura") then
                obj.Enabled = false
            end
        end
    end

    task.delay(2, function()
        for _, obj in ipairs(character:GetDescendants()) do
            if obj:GetAttribute("LimAura") then
                obj:Destroy()
            end
        end
    end)
end

local function ApplyMarketplaceClothing(assetId, clothingType)
    local ok, objects = pcall(game.GetObjects, game, "rbxassetid://" .. assetId)
    if not ok or not objects then return end
    for _, obj in ipairs(objects) do
        local clothing = obj:IsA(clothingType) and obj or obj:FindFirstChildWhichIsA(clothingType)
        if clothing then
            local existing = character:FindFirstChildOfClass(clothingType)
            if existing then existing:Destroy() end
            local newCloth = Instance.new(clothingType)
            if clothingType == "Shirt" then
                newCloth.ShirtTemplate = clothing.ShirtTemplate
            else
                newCloth.PantsTemplate = clothing.PantsTemplate
            end
            newCloth.Parent = character
            break
        end
    end
end
